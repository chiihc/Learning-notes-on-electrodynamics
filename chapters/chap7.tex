\section{守恒定律}
\subsection{电荷和能量}
连续性方程:
\begin{equation}
    \frac{\partial \rho}{\partial t}=-\boldsymbol{\nabla}\cdot\boldsymbol{J}
\end{equation}
假设我们有电荷和电流分布，在时刻t，它们产生电场$\boldsymbol{E}$和磁场$\boldsymbol{B}$。在下一时刻，$\dif t$，电荷
移动了一小段距离。问：在时间间隔$\dif t$中，作用在这些电荷上的电磁力做了多少功?
\begin{align*}
    \boldsymbol F\cdot\dif\boldsymbol{l}&=q(\boldsymbol{E}+\boldsymbol{v}\times\boldsymbol{B})\cdot\boldsymbol{v}\dif t\\
    \boldsymbol F\cdot\dif\boldsymbol{l}&=q\boldsymbol{E}\cdot\boldsymbol{v}\dif t\\
    P&=\int \boldsymbol{E}\cdot\boldsymbol{J}\dif\tau\\
    \boldsymbol{E}\cdot\boldsymbol{J}&=\frac{1}{\mu_0}\boldsymbol{E}\cdot(\boldsymbol{\nabla}\times\boldsymbol{B})
    -\varepsilon_0\boldsymbol{E}\cdot\frac{\partial \boldsymbol{E}}{\partial t}\\
    \boldsymbol{E}\cdot\boldsymbol{J}&=-\frac{1}{\mu_0}\boldsymbol{\nabla}\cdot(\boldsymbol{E}\times\boldsymbol{B})
    +\frac{1}{\mu_0}\boldsymbol{B}\cdot(\boldsymbol{\nabla}\times\boldsymbol{E})
    -\varepsilon_0\boldsymbol{E}\cdot\frac{\partial \boldsymbol{E}}{\partial t}\\
    \boldsymbol{E}\cdot\boldsymbol{J}&=-\frac{1}{\mu_0}\boldsymbol{\nabla}\cdot(\boldsymbol{E}\times\boldsymbol{B})
    -\frac{1}{\mu_0}\boldsymbol{B}\cdot\frac{\partial \boldsymbol{B}}{\partial t}
    -\varepsilon_0\boldsymbol{E}\cdot\frac{\partial \boldsymbol{E}}{\partial t}\\
    \boldsymbol{E}\cdot\boldsymbol{J}&=-\frac{1}{\mu_0}\boldsymbol{\nabla}\cdot(\boldsymbol{E}\times\boldsymbol{B})
    -\frac{1}{2\mu_0}\frac{\partial B^2}{\partial t}
    -\frac{\varepsilon_0}{2}\frac{\partial E^2}{\partial t}\\
    P&=-\frac{\dif}{\dif t}\int\frac{1}{2}\frac{\partial }{\partial t}\left(\varepsilon_0E^2+\frac{1}{\mu_0}B^2\right)\dif\tau-\frac{1}{\mu_0}\oint\boldsymbol{E}\times\boldsymbol{B}\cdot\dif\boldsymbol{a}
\end{align*}
单位时间内电磁场通过单位表面积向外传递的能量称为坡印廷矢量$\boldsymbol{S}$
\begin{equation}
    \boldsymbol{S}\equiv\frac{1}{\mu_0}\boldsymbol{E}\times\boldsymbol{B}
\end{equation}
如果我们用$u_{mech}$表示机械能能量密度，同时用$u_{em}$表示电磁场的能量密度：
\begin{equation}
    \frac{\partial u_{mech}+u_{em}}{\partial t}=-\boldsymbol{\nabla}\cdot\boldsymbol{S}
    \label{坡印廷_守恒定律}
\end{equation}

\end{align*}

\subsection{动量}
\begin{align*}
    \boldsymbol f&=\rho\boldsymbol{E}+\boldsymbol{J}\times\boldsymbol{B}\\
    \boldsymbol f&=\varepsilon_0\boldsymbol{\nabla}\cdot\boldsymbol{E}\boldsymbol{E}+\left(\frac{1}{\mu_0}\boldsymbol{\nabla}\times\boldsymbol{B}-\varepsilon_0\frac{\partial\boldsymbol{E}}{\partial t}\right)\times\boldsymbol{B}\\
    \boldsymbol f&=\varepsilon_0\boldsymbol{\nabla}\cdot\boldsymbol{E}\boldsymbol{E}+\frac{1}{\mu_0}\boldsymbol{\nabla}\times\boldsymbol{B}\times\boldsymbol{B}
    -\varepsilon_0\frac{\partial\boldsymbol{E}}{\partial t}\times\boldsymbol{B}\\
    \boldsymbol f&=\varepsilon_0\boldsymbol{\nabla}\cdot\boldsymbol{E}\boldsymbol{E}+\frac{1}{\mu_0}\boldsymbol{\nabla}\times\boldsymbol{B}\times\boldsymbol{B}
    -\varepsilon_0\frac{\partial\boldsymbol{E}\times\boldsymbol{B}}{\partial t}
    +\varepsilon_0\boldsymbol{E}\times\frac{\partial\boldsymbol{B}}{\partial t}\\
    \boldsymbol f&=\varepsilon_0\boldsymbol{\nabla}\cdot\boldsymbol{E}\boldsymbol{E}+\frac{1}{\mu_0}\boldsymbol{\nabla}\times\boldsymbol{B}\times\boldsymbol{B}
    -\varepsilon_0\frac{\partial\boldsymbol{E}\times\boldsymbol{B}}{\partial t}
    -\varepsilon_0\boldsymbol{E}\times\left(\boldsymbol{\nabla}\times\boldsymbol{E}\right)\\
    % \boldsymbol{\nabla}(\boldsymbol{E}\cdot\boldsymbol{E})&=2\boldsymbol{E}\times(\boldsymbol{\nabla}\times\boldsymbol{E})+
    % 2(\boldsymbol{E}\cdot\boldsymbol{\nabla})\boldsymbol{E}\\
    \boldsymbol f&=\varepsilon_0\left[\left(\boldsymbol{\nabla}\cdot\boldsymbol{E}\right)+\left(\boldsymbol{E}\cdot\boldsymbol{\nabla}\right)\right]\boldsymbol{E}
    +\frac{1}{\mu_0}\left[\left(\boldsymbol{\nabla}\cdot\boldsymbol{B}\right)+\left(\boldsymbol{B}\cdot\boldsymbol{\nabla}\right)\right]\boldsymbol{B}
    -\frac12\boldsymbol{\nabla}\left(\varepsilon_0E^2+\frac{1}{\mu_0}B^2\right)-\varepsilon_0\frac{\partial}{\partial t}(\boldsymbol{E}\times\boldsymbol{B})
\end{align*}
引入麦克斯韦应力张量
\begin{equation}
    T_{ij}\equiv\varepsilon_0\left(E_iE_j-\frac{1}{2}\delta_{ij}E^2\right)+\frac{1}{\mu_0}\left(B_iB_j-\frac{1}{2}\delta_{ij}B^2\right)
\end{equation}
\begin{align*}
    \boldsymbol{\nabla}\cdot\boldsymbol{T}&=\frac{\partial}{\partial x_i}\varepsilon_0\left(E_iE_j-\frac{1}{2}\delta_{ij}E^2\right)\boldsymbol{e}_j
    +\frac{\partial}{\partial x_i}\frac{1}{\mu_0}\left(B_iB_j-\frac{1}{2}\delta_{ij}B^2\right)\boldsymbol{e}_j\\
    \boldsymbol{\nabla}\cdot\boldsymbol{T}&=\varepsilon_0\frac{\partial E_iE_j}{\partial x_i}\boldsymbol{e}_j
    -\varepsilon_0\frac{1}{2}\frac{\partial E^2}{\partial x_i}\delta_{ij}\boldsymbol{e}_j
    +\frac{1}{\mu_0}\frac{\partial B_iB_j}{\partial x_i}\frac{1}{\mu_0}\boldsymbol{e}_j
    -\frac{1}{\mu_0}\frac{1}{2}\frac{\partial B^2}{\partial x_i}\delta_{ij}\boldsymbol{e}_j\\
    \boldsymbol{\nabla}\cdot\boldsymbol{T}&=\varepsilon_0\frac{\partial E_iE_j}{\partial x_i}\boldsymbol{e}_j
    -\varepsilon_0\frac{1}{2}\frac{\partial E^2}{\partial x_i}\boldsymbol{e}_i
    +\frac{1}{\mu_0}\frac{\partial B_iB_j}{\partial x_i}\boldsymbol{e}_j
    -\frac{1}{\mu_0}\frac{1}{2}\frac{\partial B^2}{\partial x_i}\boldsymbol{e}_i\\
    \boldsymbol{f}&=\boldsymbol{\nabla}\cdot\boldsymbol{T}-\varepsilon_0\mu_0\frac{\partial\boldsymbol S}{\partial t}
\end{align*}
