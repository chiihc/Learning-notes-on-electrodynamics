\section{电动力学}
\subsection{欧姆定律}
\begin{equation}
    \boldsymbol{J}=\sigma \boldsymbol{E}
\end{equation}
\mysssec{两个半径分别为$a$和$b$的同心金属球壳，被电导率为$\sigma$的弱导电材料分开\\
(a)如果它们之间保持电势差$U$，从一个球壳流向另一个的电流多大？\\
(b)球壳之间的电阻多大？}
\begin{align*}
    E&=\frac{q}{4\pi r^2}\\
    U&=\frac{q}{4\pi a}-\frac{q}{4\pi b}\\
    I&=\int J\dif S\\
    I&=\int \sigma E\dif S\\
    I&=\frac{4\pi Uab\sigma}{b-a}\\
    R&=\frac{b-a}{4\pi ab\sigma}
\end{align*}
