\mysssec{证明阿法万(Alfven)定理：在理想导电流体中(例如，自由电子气)，通过任意随流体流动的闭合回路的磁通量在所有时刻保持恒定。(磁场线，实际上，被“冻结”在流体中。)}
\begin{align*}
    \boldsymbol{J}&=\sigma(\boldsymbol{E}+\boldsymbol{v}\times\boldsymbol{B})\\
    0&=\boldsymbol{E}+\boldsymbol{v}\times\boldsymbol{B}\\
    \boldsymbol{E}&=-\boldsymbol{v}\times\boldsymbol{B}\\
    \frac{\partial \boldsymbol{B}}{\partial t}&=-\boldsymbol{\nabla}\times\boldsymbol{E}\\
    \frac{\partial \boldsymbol{B}}{\partial t}&=\boldsymbol{\nabla}\times(\boldsymbol{v}\times\boldsymbol{B})
\end{align*}
令$S$为在时刻$t$以回路($P$)为边界的面，而$S'$为在时刻$t+\dif t$以新位置($P'$)的回路为边界的面。磁通量的变化为(式中，$R$为连接$P$和$P'$的“带”)
\begin{align*}
    \dif \Phi&=\int_{S'}\boldsymbol{B}(t+\dif t)\cdot\dif \boldsymbol{a}
-\int_S\boldsymbol{B}(t)\cdot\dif \boldsymbol{a}\\
\int_{S'}\boldsymbol{B}(t+\dif t)\cdot\dif \boldsymbol{a}+\int_R\boldsymbol{B}(t+\dif t)\cdot\dif \boldsymbol{a}&=\int_S\boldsymbol{B}(t+\dif t)\cdot\dif \boldsymbol{a}\\
\dif\Phi&=\dif t\int_S\dfrac{\partial \boldsymbol{B}}{\partial t}\cdot \boldsymbol{a}-\int_R\boldsymbol{B}(t+\dif t)\cdot\dif\boldsymbol{a}\\
\dif\Phi&=\dif t\int_S\dfrac{\partial \boldsymbol{B}}{\partial t}\cdot \boldsymbol{a}
-\dif t\oint_P\boldsymbol{B}\times\boldsymbol{v}\cdot\dif\boldsymbol{l}\\
\frac{\dif\Phi}{\dif t}&=\int_S\dfrac{\partial \boldsymbol{B}}{\partial t}\cdot \boldsymbol{a}
-\oint_P\boldsymbol{B}\times\boldsymbol{v}\cdot\dif\boldsymbol{l}\\
\frac{\dif\Phi}{\dif t}&=\int_S\left[\dfrac{\partial \boldsymbol{B}}{\partial t}-\boldsymbol{\nabla}\times\boldsymbol{v}\times\boldsymbol{B}\right]\cdot \boldsymbol{a}\\
\frac{\dif\Phi}{\dif t}&=0
\end{align*}
