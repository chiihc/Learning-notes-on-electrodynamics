\subsection{一维波}
\subsubsection{波方程}
\begin{equation}
    \frac{\partial^2 f}{\partial z^2}=\frac{\partial^2 f}{v^2\partial t^2}
\end{equation}

为了便利，设节点在$z=0$处，入射波$f_1=A_1e^{i(k_1z-\omega t)},\, (z<0)$从左边人射，产生的一个反射波$f_R=A_Re^{i(-k_1z-\omega t)},\, (z<0)$沿弦1返回(故在前有负号)，以及一个透射波$f_T=A_Te^{i(k_2z-\omega t)},\, (z>0)$沿弦2继续传播。

在靠近连接点($z=0$)处，左边的位移($z=0^-$)必须与右边的位移($z=0^+$)相等，否则两弦间会有断点。如果那个结点本身的质量可忽略，$f$对$z$的一阶偏导也必须是连续的：$\left.\dfrac{\partial f}{\partial z}\right|_{0^-}=\left.\dfrac{\partial f}{\partial z}\right|_{0^+}$
否则在这个结点上会作用有一个净力，因此会产生一个无穷大的加速度。
\begin{align*}
    A_1e^{ik_1z}+A_Re^{-ik_1z}&=A_Te^{ik_2z}\\
    k_1A_1e^{ik_1z}-k_1A_Re^{-ik_1z}&=k_2A_Te^{ik_2z}
\end{align*}
解得
\begin{equation}
    A_R=\frac{k_1-k_2}{k_1+k_2}A_1, \ A_T=\frac{2k_2}{k_1+k_2}A_1
\end{equation}

\mysssec{对于两个施加有应力$T$并用质量为$m$的结点连接的弦，写出其正确的边界条件。}
\begin{align*}
    A_1e^{ik_1z}+A_Re^{-ik_1z}&=A_Te^{ik_2z}\\
    T\left.\dfrac{\partial f_1}{\partial z}\right|_{0^-}-T\left.\dfrac{\partial f_2}{\partial z}\right|_{0^+}&=m\left.\dfrac{\partial^2 f_1}{\partial t^2}\right|_{0}
\end{align*}
\mysssec{证明驻波$f=A\sin(kz)\cos(kvt)$满足波方程.把它表示成一个向左和一个向右传播的两个波之和}

    A\sin(kz)\cos(kvt)&=\frac{A}{2}\sin\left(kz+kvt\right)
    +\frac{A}{2}\sin\left(kz-kvt\right)
\end{align*}

\subsection{真空中的电磁波}
\subsubsection{$\boldsymbol E$与$\boldsymbol B$的波方程}
\begin{align}
    \boldsymbol{\nabla}\times\left(\boldsymbol{\nabla}\times\boldsymbol{E}\right)
    &=\boldsymbol{\nabla}\left(\boldsymbol{\nabla}\cdot\boldsymbol{E}\right)-\nabla^2\boldsymbol{E}\nonumber\\
    \boldsymbol{\nabla}\times\left(-\frac{\partial \boldsymbol{B}}{\partial t}\right)
    &=\boldsymbol{\nabla}0-\nabla^2\boldsymbol{E}\nonumber\\
    -\frac{\partial \boldsymbol{\nabla}\times\boldsymbol{B}}{\partial t}
    &=-\nabla^2\boldsymbol{E}\nonumber\\
    \mu_0\varepsilon_0\frac{\partial \boldsymbol{E}}{\partial t}
    &=\nabla^2\boldsymbol{E}
\end{align}
同理可得
\begin{equation}
    \mu_0\varepsilon_0\dfrac{\partial \boldsymbol{B}}{\partial t}=\nabla^2\boldsymbol{B}
\end{equation}
解得
\begin{equation}
\left.\begin{aligned}
\widetilde{\boldsymbol{E}} &= \widetilde{E}e^{i(\boldsymbol{k}\cdot\boldsymbol{r}-\omega t)}\boldsymbol{e}_n\\
\widetilde{\boldsymbol{B}} &= \frac{1}{c}\widetilde{E}e^{i(\boldsymbol{k}\cdot\boldsymbol{r}-\omega t)}\left(\boldsymbol{e}_k\times\boldsymbol{e}_n\right)=\frac{1}{c}\boldsymbol{e}_k\times\widetilde{\boldsymbol{E}}
\end{aligned}\right\}
\label{单色平面波}
\end{equation}
\subsubsection{电磁波的能量与动量}
$u=\dfrac{1}{2}\left(\varepsilon_0E^2+\dfrac{1}{\mu_0}B^2\right)$对于单色平面波$B^2=\dfrac{1}{c^2}E^2=\mu_0\varepsilon_0E^2$故电场和磁场的贡献相等
\begin{equation}
    u=\varepsilon_0E^2=\varepsilon_0E_0^2\cos^2(kz-\omega t+\delta)
\end{equation}
\begin{equation}
    \boldsymbol{S}=\frac{1}{\mu_0}\boldsymbol{E}\times\boldsymbol{B}
    =c\varepsilon_0E_0^2\cos^2(kz-\omega t+\delta)
    =cu\boldsymbol{k}
\end{equation}
动量
\begin{align*}
    \boldsymbol{\wp}=\frac{1}{c^2}\boldsymbol{S}=\frac{1}{c}u\boldsymbol{k}
\end{align*}
用括号$\langle \rangle$表示对一个周期的(时间）平均。通过单位面积传播的电磁波平均功率称为电磁波的强度：
\begin{equation}
    \langle u\rangle=\frac{1}{2}\varepsilon_0E_0^2,\, 
    \langle \boldsymbol{S}\rangle=\frac{1}{2}c\varepsilon_0E_0^2\boldsymbol{k},\, 
    \langle \boldsymbol{\wp}\rangle=\frac{1}{2c}\varepsilon_0E_0^2\boldsymbol{k}
\end{equation}
当光照射在一个物体上并被完全吸收时，光把它的动量传递给物体表面。在时间$\Delta t$内，传递的动量为$\Delta p=\langle \boldsymbol{\wp}\rangle Ac\Delta t$，故辐射压(单位面积上的平均力）是
\begin{equation}
    P=\frac{\Delta p}{A\Delta t}=\frac{1}{2}\varepsilon_0E_0^2=\frac{I}{c}
\end{equation}
对于一个完全反射的面，压强为上式的2倍。

\subsection{物质中的电磁波}
\subsubsection{在线性介质中的传播}

在物质中，在没有自由电荷和电流的区域内，麦克斯韦方程变为
\begin{equation}
\left.\begin{aligned}
\text{(i)} \quad \nabla \cdot \boldsymbol{D} &= 0, & \text{(iii)} \quad \nabla \times \boldsymbol{E} &= -\frac{\partial \boldsymbol{B}}{\partial t} \\
\text{(ii)} \quad \nabla \cdot \boldsymbol{B} &= 0, & \text{(iv)} \quad \nabla \times \boldsymbol{H} &= \frac{\partial \boldsymbol{D}}{\partial t}
\end{aligned}\right\}
\end{equation}

\noindent 如果介质是线性的，
\begin{equation}
\boldsymbol{D} = \varepsilon \boldsymbol{E}, \quad \boldsymbol{H} = \frac{1}{\mu} \boldsymbol{B}
\end{equation}
和均匀的(即 $\varepsilon$ 和 $\mu$ 不随位置变化），麦克斯韦方程简化为
\begin{equation}
\left.\begin{aligned}
\text{(i)} \quad \nabla \cdot \boldsymbol{E} &= 0, & \text{(iii)} \quad \nabla \times \boldsymbol{E} &= -\frac{\partial \boldsymbol{B}}{\partial t} \\
\text{(ii)} \quad \nabla \cdot \boldsymbol{B} &= 0, & \text{(iv)} \quad \nabla \times \boldsymbol{B} &= \varepsilon\mu \frac{\partial \boldsymbol{E}}{\partial t}
\end{aligned}\right\}
\end{equation}

\noindent 这与真空中的形式不同之处仅在于 $\mu_0\varepsilon_0$ 被 $\mu\varepsilon$ 取代。显然，电磁波通过线性均匀介质时以速度
\begin{equation}
v = \frac{1}{\sqrt{\varepsilon\mu}}
\end{equation}
传播。式中
\begin{equation}
n \equiv \sqrt{\frac{\varepsilon\mu}{\varepsilon_0\mu_0}}
\end{equation}
是物体的折射率。对大多数材料，$\mu$ 非常接近于 $\mu_0$，故
\begin{equation}
n \approx  \sqrt{\varepsilon_r}
\end{equation}
式中，$\varepsilon_r$ 是相对介电常数。因为 $\varepsilon_r$ 总是大于 1，光在物质中的传播速度稍慢。

进行简单的变换 $\varepsilon_0 \rightarrow \varepsilon, \mu_0 \rightarrow \mu$ 及 $c \rightarrow v$，以前得到的结果继续适用。能量密度为
\begin{equation}
u = \frac{1}{2}\left( \varepsilon E^2 + \frac{1}{\mu} B^2 \right)
\end{equation}
坡印廷矢量为
\begin{equation}
\boldsymbol{S} = \frac{1}{\mu}(\boldsymbol{E} \times \boldsymbol{B})
\end{equation}
对于单色平面波，角频率和波数有关系 $\omega = kv$，$\boldsymbol{B}$ 的振幅是 $\boldsymbol{E}$ 的振幅的 $1/v$，波的强度是
\begin{equation}
I = \frac{1}{2} \varepsilon v E_0^2
\end{equation}
\subsubsection{垂直入射时的反射与透射}
假设$xy$平面为两线性介质的界面，角频率为$\omega$，沿$z$方向传播和沿$x$方向偏振的平面波从左边入射到界面
\begin{equation}
\left.\begin{aligned}
\widetilde{\boldsymbol{E}}_1 &= \widetilde{E}_{01}e^{i(\boldsymbol k_1\cdot\boldsymbol z-\omega t)}\boldsymbol{e}_x\\
\widetilde{\boldsymbol{B}}_1 &= \frac{1}{v_1}\widetilde{E}_{01}e^{i(\boldsymbol k_1\cdot\boldsymbol z-\omega t)}\boldsymbol{e}_y
\end{aligned}\right\}
\end{equation}
它产生一个反射波
\begin{equation}
\left.\begin{aligned}
\widetilde{\boldsymbol{E}}_R &= \widetilde{E}_{0R}e^{i(-\boldsymbol k_1\cdot\boldsymbol z-\omega t)}\boldsymbol{e}_x\\
\widetilde{\boldsymbol{B}}_R &= -\frac{1}{v_1}\widetilde{E}_{0R}e^{i(-\boldsymbol k_1\cdot\boldsymbol z-\omega t)}\boldsymbol{e}_y
\end{aligned}\right\}
\end{equation}
反射波向左折回介质(1）中，$\widetilde{\boldsymbol{B}}_R$中的负号是式\ref{单色平面波}的要求—或者是由坡印廷矢量指向传播方向的事实得出的；及一个透射波
\begin{equation}
\left.\begin{aligned}
\widetilde{\boldsymbol{E}}_T &= \widetilde{E}_{0T}e^{i(\boldsymbol k_2\cdot\boldsymbol z-\omega t)}\boldsymbol{e}_x\\
\widetilde{\boldsymbol{B}}_T &= \frac{1}{v_2}\widetilde{E}_{0T}e^{i(\boldsymbol k_2\cdot\boldsymbol z-\omega t)}\boldsymbol{e}_y
\end{aligned}\right\}
\end{equation}
透射波继续在介质(2）中向右传播。
\begin{align*}
\left.\begin{aligned}
\widetilde{\boldsymbol{E}}_1+\widetilde{\boldsymbol{E}}_R&= \widetilde{\boldsymbol{E}}_T\\
\frac{1}{\mu_1}\widetilde{\boldsymbol{B}}_1+\frac{1}{\mu_1}\widetilde{\boldsymbol{B}}_R&= \frac{1}{\mu_2}\widetilde{\boldsymbol{B}}_T
\end{aligned}\right\}
\end{align*}
设$\beta=\dfrac{\mu_1v_1}{\mu_2v_2}=\dfrac{\mu_1n_2}{\mu_2n_1}$,\, 解得
\begin{equation}
    \widetilde{E}_{0R}=\frac{1-\beta}{1+\beta}\widetilde{E}_{01}, \ \widetilde{E}_{0T}=\frac{2}{1+\beta}\widetilde{E}_{01}
\end{equation}
如果介质的磁导率$\mu$与真空中的接近,\,则
\begin{equation}
    \widetilde{E}_{0R}=\frac{v_2-v_1}{v_2+v_1}\widetilde{E}_{01}, \ \widetilde{E}_{0T}=\frac{2v_2}{v_2-v_1}\widetilde{E}_{01}
\end{equation}
反射波和入射波强度比为
\begin{equation}
    R=\frac{I_R}{I_1}=\frac{\varepsilon_1v_1E_{0R}^2}{\varepsilon_1v_1E_{01}^2}=\left(\frac{n_1-n_2}{n_1+n_2}\right)^2
\end{equation}
透射波和入射波强度比为
\begin{equation}
    T=\frac{I_T}{I_1}=\frac{\varepsilon_2v_2E_{0T}^2}{\varepsilon_1v_1E_{01}^2}=\frac{4n_1n_2}{\left(n_1+n_2\right)^2}
\end{equation}
\subsubsection{倾斜入射时的反射与透射}

设$Ae^{iax} +Be^{ibx}=Ce^{icx}$,\, $A,B,C,a,b,c$对所有$x$都是非零常数
设$d$为满足$d>0$的任意常数
\begin{align*}
    Ae^{iax} +Be^{ibx}&=Ce^{icx}\\
    Ae^{iax-x} +Be^{ibx-x}&=Ce^{icx-x}\\
    \int_{0}^{\infty}Ae^{iax-dx} +Be^{ibx-dx}\dif x&=\int_{0}^{\infty}Ce^{icx-dx}\dif x\\
    \left.\frac{A}{ia-d}e^{iax-x} +\frac{B}{ib-d}e^{ibx-x}\right|_0^\infty&=\left.\frac{C}{ic-d}e^{icx-x}\right|_0^\infty\\
    \frac{A}{ia-d}+\frac{B}{ib-d}&=\frac{C}{ic-d}\\
\end{align*}
所以$a=b=c,\, A+B=C$


\begin{align*}
    \widetilde{E}_{01}e^{i(\boldsymbol k_1\cdot\boldsymbol r-\omega t)}+\widetilde{E}_{0R}e^{i(\boldsymbol k_R\cdot\boldsymbol r-\omega t)}=\widetilde{E}_{0T}e^{i(\boldsymbol k_T\cdot\boldsymbol r-\omega t)}
\end{align*}
显然当$z=0$时
\begin{align}
    \boldsymbol k_1\cdot\boldsymbol r
    &=\boldsymbol k_R\cdot\boldsymbol r
    =\boldsymbol k_T\cdot\boldsymbol r\\
    xk_{1x}+yk_{1y}
    &=xk_{Rx}+yk_{Ry}
    =xk_{Tx}+yk_{Ty}
\end{align}
改变坐标轴的方向使$\boldsymbol{k}_1$，在$xz$平面内
\begin{align}
    xk_{1x}
    =xk_{Rx}
    =xk_{Tx}\\
    k_{1x}
    =k_{Rx}
    =k_{Tx}\\
    \frac{\omega}{v_{1x}}
    =\frac{\omega}{v_{Rx}}
    =\frac{\omega}{v_{Tx}}
\end{align}

结论:
第一定律:入射、反射和透射波矢量在同一平面内(称为入射面)，入射面的法线也在这平面内。
\begin{equation}
    k_1\sin\theta_1=k_R\sin\theta_R=k_T\sin\theta_T
\end{equation}

第二定律:入射角等于反射角
\begin{equation}
    \theta_1=\theta_R
\end{equation}

第三定律:
\begin{equation}
    \frac{\sin\theta_T}{\sin\theta_1}=\frac{n_1}{n_2}
\end{equation}

边界条件:
\begin{align*}
\left.\begin{aligned}
\varepsilon_1\left(\widetilde{\boldsymbol{E}}_{01}+\widetilde{\boldsymbol{E}}_{0R}\right)_z&= \varepsilon_2\left(\widetilde{\boldsymbol{E}}_{0T}\right)_z\\
\left(\widetilde{\boldsymbol{B}}_{01}+\widetilde{\boldsymbol{B}}_{0R}\right)_z&= \left(\widetilde{\boldsymbol{B}}_{0T}\right)_z\\
\left(\widetilde{\boldsymbol{E}}_{01}+\widetilde{\boldsymbol{E}}_{0R}\right)_{x,y}&= \left(\widetilde{\boldsymbol{E}}_{0T}\right)_{x,y}\\
\frac{1}{\mu_1}\left(\widetilde{\boldsymbol{B}}_{01}+\widetilde{\boldsymbol{B}}_{0R}\right)_{x,y}&= \frac{1}{\mu_2}\left(\widetilde{\boldsymbol{B}}_{0T}\right)_{x,y}
\end{aligned}\right\}
\end{align*}

假设入射波的电场的偏振方向平行于入射面($xz$平面)
\begin{align*}
\left.\begin{aligned}
\varepsilon_1\left(\widetilde{{E}}_{01}\sin\theta_1-\widetilde{{E}}_{0R}\sin\theta_1\right)&= \varepsilon_2\widetilde{{E}}_{0T}\sin\theta_T\\
\left(\widetilde{\boldsymbol{B}}_{01}+\widetilde{\boldsymbol{B}}_{0R}\right)_z&=0\\
\widetilde{{E}}_{01}\cos\theta_1+\widetilde{{E}}_{0R}\cos\theta_1&= \widetilde{{E}}_{0T}\cos\theta_T\\
\widetilde{E}_{01}-\widetilde{E}_{0R}&= \beta\widetilde{E}_{0T}
\end{aligned}\right\}
\end{align*}
定义$\alpha=\dfrac{\cos\theta_T}{\cos\theta_1}$,\, 解得
\begin{equation}
    \widetilde{E}_{0R}=\frac{\alpha-\beta}{\alpha+\beta}\widetilde{E}_{01},\, \widetilde{E}_{0T}=\frac{2}{\alpha+\beta}\widetilde{E}_{01}
\end{equation}
它们称为偏振方向在入射面内的菲涅尔方程。(有另外两个菲涅尔方程，对应反射和透射波垂直于入射平面的情形。)
\mysssec{分析偏振方向垂直于入射面的情形}
\begin{align*}
\left.\begin{aligned}
\widetilde{B}_{01}+\widetilde{B}_{0R}&=\frac{\sin\theta_T}{\sin\theta_1} \widetilde{B}_{0T}\\
\widetilde{{E}}_{01}+\widetilde{{E}}_{0R}&= \widetilde{{E}}_{0T}\\
\widetilde{E}_{01}-\widetilde{E}_{0R}&=\alpha\beta\widetilde{E}_{0T}
\end{aligned}\right\}
\end{align*}
解得
\begin{equation}
    \widetilde{E}_{0R}=\frac{1-\alpha\beta}{1+\alpha\beta}\widetilde{E}_{01},\, \widetilde{E}_{0T}=\frac{2}{1+\alpha\beta}\widetilde{E}_{01}
\end{equation}

\subsubsection{导体中的电磁波}
等到任何聚集的自由电荷消失。
\begin{equation}
\left.\begin{aligned}
\text{(i)} \quad \nabla \cdot \boldsymbol{E} &= 0, & \text{(iii)} \quad \nabla \times \boldsymbol{E} &= -\frac{\partial \boldsymbol{B}}{\partial t} \\
\text{(ii)} \quad \nabla \cdot \boldsymbol{B} &= 0, & \text{(iv)} \quad \nabla \times \boldsymbol{B} &= \varepsilon\mu \frac{\partial \boldsymbol{E}}{\partial t}+\mu\sigma\boldsymbol{E}

\end{equation}

同前一样，取旋度，我们得到了$\boldsymbol E$和$\boldsymbol B$的在导体介质中修正的波方程，解得
\begin{align}
    \widetilde{\boldsymbol{E}}&= \widetilde{E}_{0}e^{i(k_1 z-\omega t)-\kappa z}\boldsymbol{e}_x\\
    \frac{\partial\widetilde{\boldsymbol{B}}}{\partial t}&=-\boldsymbol{\nabla}\times\boldsymbol{E}\\
    \frac{\partial\widetilde{\boldsymbol{B}}}{\partial t}&=-\boldsymbol{\nabla}\times\widetilde{E}_{0}e^{i(k_1 z-\omega t)-\kappa z}\boldsymbol{e}_x\\
    \frac{\partial\widetilde{\boldsymbol{B}}}{\partial t}&=-(ik_1-\kappa)\widetilde{E}_{0}e^{i(k_1 z-\omega t)-\kappa z}\boldsymbol{e}_y\\
\widetilde{\boldsymbol{B}}_1&=\frac{k_1+i\kappa}{\omega}\widetilde{E}_{0}e^{i(k_1 z-\omega t)-\kappa z}\boldsymbol{e}_y
\end{align}
式中
\begin{equation}
    k\equiv\omega\sqrt{\frac{\mu\varepsilon}{2}}\left[\sqrt{1+\left(\frac{\sigma}{\varepsilon\omega}\right)^2}+1\right]^\frac{1}{2},\, \kappa\equiv\omega\sqrt{\frac{\mu\varepsilon}{2}}\left[\sqrt{1+\left(\frac{\sigma}{\varepsilon\omega}\right)^2}-1\right]^\frac{1}{2}
\end{equation}
振幅减小到$1/e$时(约三分之一)对应的距离称为趋肤深度$d=\dfrac{1}{\kappa}$

如同任何复数，$k+i\kappa$可用模和相位表示$k+i\kappa=Ke^{i\Phi},\, K\equiv \omega\sqrt{\varepsilon\mu\sqrt{1+\left(\dfrac{\sigma}{\varepsilon\mu}\right)^2}}$
电场与磁场不再同相
\begin{equation}
    B_0e^{i\delta_B}=\frac{Ke^{i\Phi}}{\omega}E_0e^{i\delta_E}
\end{equation}

\subsubsection{介电常数对频率的依赖}

% 在静电学中，介电常数 $\varepsilon$ 通常被视为一个常数。然而，当电磁波在介质中传播时，交变电场会驱动介质中的电荷运动。由于电荷（电子、离子、分子）具有质量和惯性，它们的响应无法完美同步于高频交变电场。因此，介电常数必然依赖于交变电场的角频率 $\omega$，这种现象称为\textbf{介电色散（Dielectric Dispersion）}。

% 为了全面描述能量的高效存储（色散）与损耗（吸收），我们引入\textbf{复介电常数（Complex Dielectric Function）}：
% \[
% \widetilde{\varepsilon}(\omega) = \varepsilon_1(\omega) + i\varepsilon_2(\omega)
% \]

% 介质对电场的总极化强度 $\mathbf{P}$ 是多种微观极化机制叠加的结果。每种机制都有其特定的响应时间（弛豫时间或共振频率）：


外加一个角频率为$\omega$，偏振方向沿$x$方向的电磁波，电子受到某些阻尼作用，阻尼方向与速度必须相反，使它与速度成正比较易求解，且处于某个势场中，运动方程为
\begin{equation}
    m\frac{\dif^2\widetilde{x}}{\dif t^2} + m\gamma\frac{\dif \widetilde{x}}{\dif t} + m\omega_0^2\widetilde{x}= eE_0 e^{-i\omega t}
\end{equation}
解得
\begin{equation}
    \widetilde{x}=\frac{q}{m\omega_0^2-m\omega^2-im\gamma\omega}E_0
\end{equation}

设稳态解具有形式 $\mathbf{x}(t) = \mathbf{x}_0 e^{-i\omega t}$，代入上式得到电荷的位移振幅：
\[
\mathbf{x}_0 = \frac{e/m}{\omega_0^2 - \omega^2 - i\gamma\omega}\mathbf{E}_0
\]
单个电荷产生的感应偶极矩为 $\mathbf{p} = e\mathbf{x}$。若单位体积内的振子数为 $N$，则宏观极化强度 $\mathbf{P}$ 为：
\[
\mathbf{P} = N\mathbf{p} = \frac{Ne^2/m}{\omega_0^2 - \omega^2 - i\gamma\omega}\mathbf{E}_0 e^{-i\omega t}
\]


根据宏观电动力学关系 $\mathbf{D} = \varepsilon_0\mathbf{E} + \mathbf{P} = \widetilde{\varepsilon}\mathbf{E}$，可得复相对介电常数 $\widetilde{\varepsilon}_r(\omega) = \frac{\widetilde{\varepsilon}(\omega)}{\varepsilon_0}$：
\[
\widetilde{\varepsilon}_r(\omega) = 1 + \frac{Ne^2}{m\varepsilon_0} \frac{1}{\omega_0^2 - \omega^2 - i\gamma\omega}
\]
定义\textbf{等离子体频率（Plasma Frequency）} $\omega_p = \sqrt{\frac{Ne^2}{m\varepsilon_0}}$，上式写为标准形式：
\[
\widetilde{\varepsilon}_r(\omega) = 1 + \frac{\omega_p^2}{\omega_0^2 - \omega^2 - i\gamma\omega}
\]

将复介电常数展开为实部和虚部 $\widetilde{\varepsilon}_r = \varepsilon'_r + i\varepsilon''_r$：
\[
\varepsilon'_r(\omega) = 1 + \frac{\omega_p^2 (\omega_0^2 - \omega^2)}{(\omega_0^2 - \omega^2)^2 + \gamma^2\omega^2}
\]
\[
\varepsilon''_r(\omega) = \frac{\omega_p^2 \gamma\omega}{(\omega_0^2 - \omega^2)^2 + \gamma^2\omega^2}
\]

\begin{itemize}
    \item \textbf{正常色散（Normal Dispersion）}：在远离共振频率 $\omega_0$ 的区域，$\varepsilon'_r$ 随 $\omega$ 的增加而增加（对应折射率 $n$ 随频率升高而变大，如蓝光的折射率大于红光）。
    \item \textbf{反常色散（Anomalous Dispersion）}：在共振频率附近（$\omega \approx \omega_0$），$\varepsilon'_r$ 随 $\omega$ 增加而剧烈下降。
\end{itemize}

\begin{itemize}
    \item $\varepsilon''_r(\omega)$ 严格正相关于介质对电磁波的\textbf{吸收（Absorption）}。
    \item 在共振点 $\omega = \omega_0$ 处，$\varepsilon''_r$ 达到极大值，形成一个强吸收峰（Absorptive Peak），此时电磁波能量被高效转化为介质的内能。
\end{itemize}

当考虑金属中的\textbf{自由电子}时，恢复力消失，即固有共振频率 $\omega_0 = 0$。此时洛伦兹模型退化为德鲁德模型：
\[
\widetilde{\varepsilon}_r(\omega) = 1 - \frac{\omega_p^2}{\omega^2 + i\gamma\omega}
\]
若在光学和高频频段忽略碰撞阻尼（$\gamma \ll \omega$）：
\[
\widetilde{\varepsilon}_r(\omega) \approx 1 - \frac{\omega_p^2}{\omega^2}
\]
\begin{itemize}
    \item \textbf{当 $\omega < \omega_p$ 时}：$\widetilde{\varepsilon}_r$ 为\textbf{负实数}。这导致波数变成纯虚数，波在金属内部发生全反射（消逝波），无法向内传播。
    \item \textbf{当 $\omega > \omega_p$ 时}：$\widetilde{\varepsilon}_r > 0$，金属在紫外等高频段会突然变得透明。
\end{itemize}
